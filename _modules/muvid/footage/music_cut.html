muvid.footage.music_cut

Source code for muvid.footage.music_cut

"""Cut footage to the music — the montage half of a music video.

A music video in muvid has always assumed its footage is a RECORDING OF THE SONG — a
concert, a dance, a lip-sync — so every clip is *synced*: placed on the song by
listening to its own soundtrack (:mod:`muvid.footage.align`) and shown at exactly the
moment it was filmed. That is the right answer for that footage and the wrong one for
the other common case, which is the one phone galleries sell as "memories": a day out
filmed in short clips and photos, plus a song that was never playing. There is nothing
to sync to, so the aligner rightly cannot place those clips, and an edit built only from
synced clips is mostly black.

This module is the other answer. Footage that does not contain the song is cut TO the
music instead of synced WITH it:

* **where the cuts fall** is the montage planner's decision
  (:func:`muvid.montage.plan.plan_montage`): every cut on a beat, bar or section
  boundary, denser in loud sections, with the reuse policy that lets a few clips carry a
  whole song and keeps a revisited clip from showing the same stretch twice;
* **which stretch of a video each cut shows** is decided here, by the picture: the
  stretch whose visual hits (:func:`muvid.footage.beats.activity_signal` — the onsets of
  picture change, camera moves included) land on the beats inside the cut, and whose
  liveliness suits the section's loudness. That is the alignment phone "memories" do
  not do;
* **how a still moves** is a named camera look (push, pull, pan) toward the photo's
  salient region (``burns.salient_box`` when ``burns`` is installed).

The output is ordinary :class:`~muvid.footage.edl.EdlEntry` cuts with
:attr:`~muvid.footage.edl.EdlEntry.source_in` set — *free* cuts — so a montage is an
edit like any other: saved, changed cut by cut, rendered by the same assembler over the
clean song. Synced and free cuts mix in one edit: :func:`fill_spans` fills only the
spans it is given, which the service uses to put free cuts in the gaps a synced edit
leaves.

Pure planning: no file is written; the measurements (song analysis, picture-change
envelopes, salient boxes) come in through keyword seams with working defaults.
"""

#: Pick the cutting style from the song (see :func:`pick_archetype`).
ARCHETYPE_AUTO = "auto"
#: Below this tempo the auto choice is the slow dissolve, at or above it hard cuts.
SLOW_TEMPO_BPM = 92.0
#: The montage archetypes a free cut can express: one picture per slot. ``grid`` tiles
#: several pictures in one frame, which an edit (one clip per span) cannot hold.
SINGLE_PICTURE_ARCHETYPES = ("beat_cut", "ballad_dissolve", "stop_motion")
#: A piece of a cut shorter than this (seconds) is folded into its neighbour.
MIN_PIECE_S = 0.4
#: The tolerance for comparing two times this module COPIED from one another (a piece's
#: end and the next one's start, a beat and a bound read off it): float noise only, so
#: finer than the gate's ``edl.COINCIDENT_S``, which is what a cut's own length is
#: judged by (muvid#210).
_SAME_FLOAT = 1e-6
#: Weights of the source-in score: beats hit, liveliness matched to loudness, reuse,
#: and focus (log of the stretch's sharpness over the video's own, so a whip-pan blur —
#: which changes a lot and shows nothing — loses to a lively stretch in focus).
W_HITS, W_ENERGY, W_REUSE, W_SHARP = 1.0, 0.35, 2.0, 0.6
#: How far a still's camera move goes (zoom factor; the named looks cap it at 1.15).
STILL_ZOOM = 1.12
#: The camera moves a still cycles through, by how often the still has been used, when
#: the planner left it unmoved — so a returning photo moves differently.
STILL_MOVES = ("slow_push", "pan_right", "slow_pull", "pan_left")
#: The montage planner's motion vocabulary, as muvid's named looks.
MOTION_TO_LOOK = {
    "zoom_in": "slow_push",
    "zoom_out": "slow_pull",
    "pan_left": "pan_left",
    "pan_right": "pan_right",
    "punch": "punch_in",
}
#: Looks that take an anchor (where to zoom toward).
_ANCHORED_LOOKS = ("slow_push", "slow_pull", "punch_in")
#: How a picture meets a canvas of another shape: ``cover`` fills the frame (cropping
#: the overflow, around the photo's focal point — what phone "memories" do), ``contain``
#: shows all of it with bars (what a synced recording of the song gets).
FITS = ("cover", "contain")
#: Aspect ratios closer than this are treated as equal (no crop for 2 %).
_ASPECT_TOLERANCE = 0.02
STILL = "still"
VIDEO = "video"
#: How the pictures are ordered across the song. ``taken``: by capture time, so a day
#: reads as it happened (muvid#137) — upload order where a file has no capture time;
#: ``planner``: the montage planner's own reuse policy (least used first, a golden-ratio
#: stride, the strongest held for the finale), which ignores chronology.
ORDERS = ("taken", "planner")
DEFAULT_ORDER = "taken"

[docs]
    @classmethod
    def from_signals(cls, record: Mapping) -> "Optional[Envelope]":
        """From :func:`muvid.footage.beats.activity_signal`'s ``signals`` mapping."""
        from muvid.footage.beats import ACTIVITY, ACTIVITY_HITS, SHARPNESS

        a, h, f = record.get(ACTIVITY), record.get(ACTIVITY_HITS), record.get(SHARPNESS)
        if not a or not h or not a.get("n"):
            return None

        def arr(sig):
            return np.array(
                [np.nan if v is None else v for v in sig["values"]], dtype=float
            )

        return cls(
            hop_s=float(a["hop_s"]),
            activity=arr(a),
            hits=arr(h),
            sharpness=arr(f) if f and f.get("n") else None,
        )

def _room(cut, *, clip_duration: float, next_transition=None) -> tuple[float, float]:
    """The in-points a free ``cut`` may take in a clip of ``clip_duration`` seconds:
    :func:`muvid.footage.edl.in_point_bounds`, keeping :data:`EDGE_S` from each end
    where it fits (muvid#192).

    A clip too short for the span AND both blends still gets a room — the in-point
    that keeps the SPAN inside the clip (the gate's containment rule) — and
    :func:`_feasible_blends` then drops whichever blend no longer fits. The old rule
    returned the full lead here, which could run the span off the clip's end. That
    fallback ignores ``slip_s``, which the planner never sets.

    >>> from muvid.footage.edl import EdlEntry, Transition
    >>> fade = Transition(duration_s=1.0, curve="fade")
    >>> _room(EdlEntry(0.0, 2.0, "c", transition=fade), clip_duration=2.25)
    (0.25, 0.25)
    """
    from muvid.footage.edl import blend_lead, in_point_bounds

    room = in_point_bounds(
        cut,
        clip_duration_s=clip_duration,
        next_transition=next_transition,
        edge_s=EDGE_S,
    )
    if room is not None:
        return room
    lead = blend_lead(cut.transition) if cut.transition is not None else 0.0
    length = cut.song_end - cut.song_start
    at = min(lead, max(0.0, clip_duration - length))
    return at, at

def choose_source_in(
    env: "Optional[Envelope]",
    *,
    clip_duration: float,
    length: float,
    beat_offsets: Sequence[float] = (),
    used: Sequence[tuple[float, float]] = (),
    energy: float = 0.0,
    room: "Optional[tuple[float, float]]" = None,
    ordinal: int = 0,
) -> tuple[float, float]:
    """Where in a video a cut of ``length`` seconds should start: ``(source_in, fit)``.

    Scores every in-point on the envelope's grid within ``room`` — the legal in-points,
    :func:`_room` (a blend-less cut's when omitted) — by three terms:

    * **hits on the beat** — the mean picture-change hit at each ``beat_offsets`` (beats
      inside the cut, in seconds from its start), relative to the video's own mean hit:
      above 1 means the picture changes ON the beat more than it does anywhere;
    * **liveliness matched to loudness** — ``energy`` in [-1, 1] (quiet to loud
      section) times how lively the stretch is relative to the whole video;
    * **reuse** — the fraction of the stretch already shown by another cut, penalised.

    ``fit`` is the hits term at the chosen point (0 when there are no inner beats or no
    envelope). Without an envelope it falls back to a golden-ratio stride over the
    room, the montage planner's own blind rule, with reuse still avoided.

    >>> hits = np.zeros(100); hits[[20, 30]] = 1.0   # changes at 2.0 s and 3.0 s
    >>> env = Envelope(hop_s=0.1, activity=np.full(100, 0.1), hits=hits)
    >>> s, fit = choose_source_in(env, clip_duration=10.0, length=2.0, beat_offsets=[0.5, 1.5])
    >>> round(s, 2), fit > 1
    (1.5, True)
    """
    if room is None:
        from muvid.footage.edl import EdlEntry

        room = _room(EdlEntry(0.0, length, ""), clip_duration=clip_duration)
    room_lo, room_hi = room
    if env is None or env.hits.size < 2:
        span = max(0.0, room_hi - room_lo)
        candidates = np.linspace(room_lo, room_hi, max(2, int(span / 0.25) + 1))
        golden = room_lo + ((ordinal * 0.6180339887498949) % 1.0) * span
        overlap = np.array([_overlap(s, s + length, used) for s in candidates])
        best = candidates[np.argmin(overlap * 1e3 + np.abs(candidates - golden))]
        return math.floor(max(0.0, float(best)) * 1e4) / 1e4, 0.0

    hop = env.hop_s
    hits = np.nan_to_num(env.hits, nan=0.0)
    act = np.nan_to_num(env.activity, nan=0.0)
    mean_hit = float(np.mean(hits)) + 1e-9
    mean_act = float(np.mean(act)) + 1e-9
    i_lo = int(math.ceil(room_lo / hop))
    i_hi = int(math.floor(room_hi / hop))
    if i_hi < i_lo:
        i_hi = i_lo
    starts = np.arange(i_lo, i_hi + 1)
    n = hits.size
    offsets = [o for o in beat_offsets if 0.0 < o < length]

    def at(sig, idx):
        return sig[np.clip(idx, 0, n - 1)]

    if offsets:
        # a beat is "hit" by a change on its grid step, half-credited one step off
        acc = np.zeros(starts.size)
        for o in offsets:
            k = starts + int(round(o / hop))
            near = np.maximum(at(hits, k - 1), at(hits, k + 1))
            acc += np.maximum(at(hits, k), 0.5 * near)
        hit_term = acc / len(offsets) / mean_hit
    else:
        hit_term = np.zeros(starts.size)
    w = max(1, int(round(length / hop)))
    csum = np.concatenate([[0.0], np.cumsum(act)])
    lively = (csum[np.clip(starts + w, 0, n)] - csum[np.clip(starts, 0, n)]) / w
    lively_term = energy * (lively / mean_act - 1.0)
    reuse = np.array(
        [_overlap(s * hop, s * hop + length, used) / length for s in starts]
    )
    score = W_HITS * hit_term + W_ENERGY * lively_term - W_REUSE * reuse
    if env.sharpness is not None and env.sharpness.size:
        sharp = np.nan_to_num(env.sharpness, nan=0.0)
        ssum = np.concatenate([[0.0], np.cumsum(sharp)])
        m = sharp.size
        window = (ssum[np.clip(starts + w, 0, m)] - ssum[np.clip(starts, 0, m)]) / w
        mean_sharp = float(np.mean(sharp)) + 1e-9
        score = score + W_SHARP * np.clip(
            np.log((window + 1e-9) / mean_sharp), -2.0, 1.0
        )
    j = int(np.argmax(score))
    # never past the room (grid rounding), never negative, rounded DOWN for the wire
    s_in = max(0.0, min(float(starts[j] * hop), max(room_lo, room_hi)))
    return math.floor(s_in * 1e4) / 1e4, float(hit_term[j])




def _overlap(a: float, b: float, used: Sequence[tuple[float, float]]) -> float:
    return sum(max(0.0, min(b, u1) - max(a, u0)) for u0, u1 in used)

def pick_archetype(analysis) -> str:
    """The cutting style for a song: a slow dissolve below :data:`SLOW_TEMPO_BPM`, hard
    cuts on the beat otherwise.

    >>> class A: tempo_bpm = 80.0
    >>> pick_archetype(A())
    'ballad_dissolve'
    """
    return "ballad_dissolve" if analysis.tempo_bpm < SLOW_TEMPO_BPM else "beat_cut"




#: Fill a hole in the beat grid longer than this many beats (a quiet passage the beat
#: tracker went silent on) — and run the grid on to the song's ends.
GRID_HOLE_BEATS = 1.6


def extend_grid(analysis, duration: Optional[float] = None):
    """``analysis`` with its beat grid carried through the quiet passages a beat
    tracker goes silent on, and on to both ends of the song, at its own pace.

    Without this, a song whose last 25 s are a quiet outro has no beats there: the
    planner cannot cut on them, and the last picture holds to the end (measured: a
    21 s hold on a 4:24 song whose beat was last found at 3:58). Returns
    ``(analysis, note)``; ``note`` is ``None`` when nothing was added.

    >>> from muvid.montage.analysis import Analysis
    >>> a = Analysis(duration=4.6, tempo_bpm=120.0, beats=(1.0, 1.5, 2.0, 2.5, 3.0),
    ...              downbeats=(1.0, 3.0))
    >>> b, note = extend_grid(a)
    >>> b.beats[:3], b.beats[-2:], b.downbeats, note is not None
    ((0.5, 1.0, 1.5), (3.5, 4.0), (1.0, 3.0), True)
    """
    from dataclasses import replace

    beats = list(analysis.beats)
    dur = float(duration if duration is not None else analysis.duration)
    if len(beats) < 4:
        return analysis, None
    period = float(np.median(np.diff(beats)))
    if not period > 0:
        return analysis, None
    out = []
    filled = []
    for a, b in zip(beats, beats[1:]):
        out.append(a)
        if b - a > GRID_HOLE_BEATS * period:
            n = int(round((b - a) / period))
            step = (b - a) / n
            out += [a + k * step for k in range(1, n)]
            filled.append((a, b))
    out.append(beats[-1])
    head = []
    t = beats[0] - period
    while t >= 0.5 * period:  # no beat crammed against 0
        head.insert(0, t)
        t -= period
    tail = []
    t = beats[-1] + period
    while t < dur - 0.5 * period:
        tail.append(t)
        t += period
    if not head and not tail and not filled:
        return analysis, None
    grid = tuple(round(x, 4) for x in head + out + tail)
    bpb = int(getattr(analysis, "beats_per_bar", 4) or 4)
    new_downs = _carried_downbeats(grid, analysis.downbeats, bpb)
    if tail:
        note = (
            f"the song's beat was last heard at {_clock(beats[-1])}; it was carried on, at "
            "the song's own pace, to the end"
        )
    elif filled:
        note = (
            "the song's beat was carried on through its quiet passages, at its own pace"
        )
    else:
        note = None  # only a lead-in before the first beat: nothing worth saying
    return replace(analysis, beats=grid, downbeats=new_downs), note


def _carried_downbeats(grid, measured, bpb: int) -> tuple:
    """The measured downbeats, kept exactly, plus every ``bpb``-th grid beat before
    the first and after each measured one where the measurement leaves a gap of more
    than a bar and a half — so a filled hole or a carried-on ending gets bars without
    relabelling a single measured one."""
    g = list(grid)
    if not g:
        return tuple(measured)
    idx = sorted(
        {min(range(len(g)), key=lambda i: abs(g[i] - d)) for d in measured}
    ) or [0]
    out = set(idx)
    for a, b in zip(idx, idx[1:] + [len(g) + bpb]):
        k = a + bpb
        while k < len(g) and b - k > bpb // 2:
            out.add(k)
            k += bpb
    k = idx[0] - bpb
    while k >= 0:
        out.add(k)
        k -= bpb
    return tuple(g[i] for i in sorted(out))


def _clock(t: float) -> str:
    """``72.41`` -> ``"1:12.4"``."""
    m, sec = divmod(max(0.0, float(t)), 60.0)
    return f"{int(m)}:{sec:04.1f}"


def _default_analysis(song_path: str):

def _file_key(path: str) -> tuple:
    """``(path, mtime_ns, size)`` — what a per-file memo is keyed on, so a replaced file
    is measured again and an unchanged one never is."""
    import os

    try:
        st = os.stat(path)
    except OSError:
        return (str(path), None, None)
    return (str(path), st.st_mtime_ns, st.st_size)


def _memo(fn):
    """Memoise a per-file measurement for the life of the process (a server answers
    many "Cut it for me"s over the same files)."""
    from functools import lru_cache, wraps

    @lru_cache(maxsize=1024)
    def cached(key, *rest):
        return fn(key[0], *rest)

    @wraps(fn)
    def wrapper(path, *rest):
        return cached(_file_key(path), *rest)

    wrapper.cache_clear = cached.cache_clear
    return wrapper


@_memo
def _salient_centre(path: str) -> tuple[float, float]:
    from burns import salient_box

    x, y, w, h = salient_box(path)
    return float(x + w / 2.0), float(y + h / 2.0)


def _default_anchor(source: FreeSource) -> tuple[float, float]:
    """Where a still's camera move should go: the centre of ``burns.salient_box`` —
    the photo's busy, detailed region, away from sky and wall — or the frame's centre
    without ``burns``."""
    try:
        return _salient_centre(source.path)
    except Exception:  # no burns, or an unreadable image: the centre, never a failure
        return 0.5, 0.5


@_memo
def display_size(path: str) -> Optional[tuple[int, int]]:
    """A picture's size as SHOWN — a phone video stored 640x360 with a -90 rotation is
    360x640 on screen, and ffmpeg rotates before any filter, so a crop must be computed
    on this size, not the stored one. ``None`` when it cannot be read."""
    from muvid.visualize.ffmpeg import probe

    try:
        streams = probe(path)["streams"]
    except Exception:
        return None
    video = next((s for s in streams if s.get("codec_type") == "video"), None)
    if not video or not video.get("width"):
        return None
    w, h = int(video["width"]), int(video["height"])
    rotation = 0.0
    for side in video.get("side_data_list") or []:
        if "rotation" in side:
            rotation = float(side["rotation"])
    rotation = rotation or float((video.get("tags") or {}).get("rotate", 0) or 0)
    if int(round(abs(rotation))) % 180 == 90:
        w, h = h, w
    return w, h


def cover_crop(
    size: Optional[tuple[int, int]],
    canvas: tuple[int, int],
    *,
    anchor: tuple[float, float] = (0.5, 0.5),
):
    """The crop window that fills ``canvas`` from a picture of ``size``, centred as near
    ``anchor`` as the picture allows; ``None`` when the shapes already match.

    >>> c = cover_crop((1920, 1080), (1080, 1920))     # landscape onto portrait
    >>> round(c.w, 3), c.h, round(c.x, 3)
    (0.316, 1.0, 0.342)
    >>> cover_crop((1080, 1920), (1080, 1920)) is None
    True
    """
    from muvid.footage.edl import CropWindow

    if not size or not size[0] or not size[1]:
        return None
    a, target = size[0] / size[1], canvas[0] / canvas[1]
    if abs(a / target - 1.0) <= _ASPECT_TOLERANCE:
        return None
    if a > target:  # wider than the canvas: keep full height, take a column
        w = target / a
        x = min(max(anchor[0] - w / 2.0, 0.0), 1.0 - w)
        return CropWindow(x=round(x, 4), y=0.0, w=round(w, 4), h=1.0)
    h = a / target
    y = min(max(anchor[1] - h / 2.0, 0.0), 1.0 - h)
    return CropWindow(x=0.0, y=round(y, 4), w=1.0, h=round(h, 4))


@_memo
def _probe_one(path: str, kind: str):
    from muvid.montage.analysis import probe_media

    (m,) = list(probe_media([path], kind=kind, start_index=0))
    return m


def _media_for(sources: Sequence[FreeSource]):
    """The montage planner's pool, one :class:`~muvid.montage.analysis.Media` per source,
    ranked by the planner's own strength measure."""
    from dataclasses import replace

    from muvid.montage.analysis import Media

    media = []
    for i, s in enumerate(sources):
        kind = "photo" if s.is_still else "clip"
        try:
            m = replace(_probe_one(s.path, kind), index=i)
        except Exception:  # unmeasurable: plan with it anyway, weakest
            m = Media(index=i, path=s.path, kind=kind, width=1, height=1, strength=0.0)
        if not s.is_still and s.duration_s is not None:
            m = Media(
                index=i,
                path=m.path,
                kind=kind,
                width=m.width,
                height=m.height,
                duration=float(s.duration_s),
                strength=m.strength,
            )
        media.append(replace(m, taken_at=s.taken_at))
    return media


def _mixed_clock_note(sources: Sequence[FreeSource]) -> str:
    """A caution when the capture times are not on one clock: a video that only carries a
    UTC ``creation_time`` is compared as is, hours from photos beside it."""
    from muvid.footage.capture_time import CREATION_TIME

    kinds = {s.taken_source for s in sources if s.taken_at is not None}
    if CREATION_TIME in kinds and len(kinds) > 1:
        return (
            "some videos only carry a UTC capture time, so they may sit hours from the "
            "photos taken at the same moment"
        )
    return ""


def _treatment(archetype: str, cut_feel: str):
    from muvid.montage import spec as spec_mod
    from muvid.montage.pipeline import build_treatment

    treatment, _notes, _src = build_treatment({"archetype": archetype})
    if cut_feel and cut_feel != treatment.direction.cut_feel:
        from dataclasses import replace

        if cut_feel not in spec_mod.CUT_FEELS:
            raise ValueError(
                f"unknown pace {cut_feel!r}; the paces are {sorted(spec_mod.CUT_FEELS)}"
            )
        treatment = replace(
            treatment, direction=replace(treatment.direction, cut_feel=cut_feel)
        )
    return treatment


def liveliness(src: FreeSource, env: "Optional[Envelope]") -> Optional[float]:
    """The default ``affect_of`` (muvid#134): how much a video's picture moves — the
    mean of its picture-change activity — or ``None`` (a still, or no envelope).
    Arousal only, of the whole clip (not per stretch: the in-point choice already
    matches a stretch's liveliness to its section), and raw — :func:`fill_spans`
    rescales it across the pool."""
    if env is None or env.activity.size == 0:
        return None
    return float(np.mean(np.nan_to_num(env.activity, nan=0.0)))


#: How far apart (relative to the larger) the pool's affects must be to tell pictures
#: apart (muvid#134): below it, two clips differ by noise — compression, low light —
#: and splitting the pool on that would only add repeats.
AFFECT_MIN_SPREAD = 0.25


def _affect_mismatch(sources, analysis, energy_at, *, affect_of, envelope):
    """``mismatch(start, end, source_index)`` — ``|arousal - loudness|`` in [0, 1] for a
    picture shown over ``[start, end)`` (muvid#134), arousal being the picture's
    ``affect_of`` value scaled over the pool (:func:`_affect_band`) and the loudness its
    section's — or ``None`` when there is nothing to match: no ``affect_of``, fewer than
    two sections with a measured loudness, or affects that do not differ by
    :data:`AFFECT_MIN_SPREAD`. A picture with no affect (or a non-finite one) sits in
    the middle."""
    measured = [s for s in analysis.sections if s.energy_db is not None]
    if affect_of is None or len(measured) < 2:
        return None
    raw = {i: affect_of(s, envelope(s)) for i, s in enumerate(sources)}
    known = {i: float(v) for i, v in raw.items() if v is not None and math.isfinite(v)}
    band = _affect_band(list(known.values()))
    if band is None:
        return None
    lo, hi = band
    arousal = {i: min(1.0, max(0.0, (v - lo) / (hi - lo))) for i, v in known.items()}

    def mismatch(start: float, end: float, index: int) -> float:
        loud = (energy_at((start + end) / 2.0) + 1.0) / 2.0
        return abs(arousal.get(index, 0.5) - loud)

    return mismatch


#: The share of the pool trimmed off EACH end before the affects are scaled
#: (muvid#234), at least one clip once the pool has more than :data:`AFFECT_TRIM_MIN_POOL`
#: — a count, not a percentile, so one wild clip is trimmed from a pool of five as from a
#: pool of fifty.
AFFECT_TRIM_SHARE = 0.1
#: The smallest pool trimmed at all: in three clips the extremes ARE the pool.
AFFECT_TRIM_MIN_POOL = 4


def _spread_enough(lo: float, hi: float) -> bool:
    return hi - lo > AFFECT_MIN_SPREAD * max(abs(hi), abs(lo), 1e-12)


def _affect_band(values: Sequence[float]) -> "Optional[tuple[float, float]]":
    """The ``(lo, hi)`` the pool's affects are scaled between (values outside clip to
    the end), or ``None`` when they do not differ enough to match on (muvid#234).

    The TRIMMED band first — :data:`AFFECT_TRIM_SHARE` of the pool off each end — so one
    wild clip no longer squashes the lively ones into misfits in every loud section, as
    the pool's min and max did. The spread guard reads that same band, so noise between
    near-identical clips is never stretched to the full scale. Where the trimmed band is
    flat (all but a few clips alike) the full range is the honest scale: the few ARE the
    lively ones.

    >>> _affect_band([.05] * 4 + [.6] * 3 + [10.])  # one wild clip in eight: trimmed off
    (0.05, 0.6)
    >>> _affect_band([.49, .5, .5, .51, 10.])        # noise + one outlier: the full range
    (0.49, 10.0)
    >>> _affect_band([.49, .5, .51]) is None         # noise alone: no affect
    True
    """
    vs = sorted(values)
    if not vs:
        return None
    k = (
        max(1, math.ceil(AFFECT_TRIM_SHARE * len(vs)))
        if len(vs) >= AFFECT_TRIM_MIN_POOL
        else 0
    )
    for lo, hi in ((vs[k], vs[-1 - k]), (vs[0], vs[-1])):
        if _spread_enough(lo, hi):
            return lo, hi
    return None


def _cut_affect_fit(entries, sources, mismatch) -> Optional[float]:
    """``1 - mismatch`` averaged over the cuts' time, or ``None`` without affect."""
    if mismatch is None or not entries:
        return None
    index = {s.clip_id: i for i, s in enumerate(sources)}
    total = good = 0.0
    for e in entries:
        length = e.song_end - e.song_start
        good += length * (
            1.0 - mismatch(e.song_start, e.song_end, index.get(e.clip_id, -1))
        )
        total += length
    return round(good / total, 3) if total else None


#: How a song's fit to the footage is scored (muvid#135), from the planner's own account
#: of cutting the footage to it, each term on a bounded scale: the pictures changing on
#: its beat (``tanh(on_beat - 1)``, measured pictures only), their feel fitting its loud
#: and quiet parts (``affect_fit`` mapped to [-1, 1]), the song left with no picture (a
#: fraction — a black hole in the video, the worst), and how many times over the footage
#: must be shown to fill it (above once, capped at 1).
SONG_FIT_WEIGHTS = {"beat": 1.0, "affect": 1.0, "uncovered": 3.0, "reuse": 1.0}
#: How many seconds of footage a photo counts for when judging repetition: shown much
#: longer than a few moves, the same photo is repetition.
STILL_FOOTAGE_S = 4.0


def song_fit(
    song_path: str,
    sources: Sequence[FreeSource],
    *,
    analysis: Any = None,
    weights: Mapping[str, float] = SONG_FIT_WEIGHTS,
    still_footage_s: float = STILL_FOOTAGE_S,
    **fill_kw,
) -> dict:
    """How well ``sources`` cut to the music of ``song_path`` (muvid#135): plan the cut
    (:func:`fill_spans`, pure — nothing is rendered) and score the plan on its own
    terms (:data:`SONG_FIT_WEIGHTS`). ``{"score", "on_beat", "affect_fit",
    "uncovered_s", "times_over", "duration_s", "n_cuts"}``; higher ``score`` fits
    better. ``on_beat`` is ``None`` when no video's picture was measured, ``affect_fit``
    when the feel could not be matched (each then counts nothing). ``fill_kw`` reaches
    :func:`fill_spans`; the order defaults to ``planner`` so the feel can be matched.
    """
    fill_kw.setdefault("order", "planner")
    cut = fill_spans(song_path, sources, analysis=analysis, **fill_kw)
    r = cut.report
    covered = sum(e.song_end - e.song_start for e in cut.entries)
    uncovered_s = float(r.get("uncovered_s", 0.0))
    duration = float(getattr(analysis, "duration", 0.0) or covered + uncovered_s)
    measured = set(r.get("measured_pictures", ()))
    beats = [v for cid, v in r.get("on_beat", {}).items() if cid in measured]
    beat = float(np.mean(beats)) if beats else None
    footage = sum(
        still_footage_s if s.is_still else float(s.duration_s or 0.0) for s in sources
    )
    times_over = covered / footage if footage else 0.0
    affect = r.get("affect_fit")
    score = (
        weights["beat"] * (math.tanh(beat - 1.0) if beat is not None else 0.0)
        + weights["affect"] * (2.0 * affect - 1.0 if affect is not None else 0.0)
        - weights["uncovered"] * (uncovered_s / duration if duration else 0.0)
        - weights["reuse"] * min(1.0, max(0.0, times_over - 1.0))
    )
    return {
        "score": round(score, 4),
        "on_beat": None if beat is None else round(beat, 3),
        "affect_fit": affect,
        "uncovered_s": round(uncovered_s, 2),
        "times_over": round(times_over, 2),
        "duration_s": round(duration, 2),
        "n_cuts": len(cut.entries),
    }


def song_fit_sentence(name: str, fit: dict) -> str:
    """One sentence a person reads about ``fit`` (:func:`song_fit`) for song ``name``."""
    parts = []
    beat = fit["on_beat"]
    if beat is not None and beat > 1.05:
        parts.append(
            f"the videos change on its beat {beat:.1f} times as often as elsewhere"
        )
    elif beat is not None and beat < 0.95:
        parts.append("the videos change off its beat more than on it")
    if fit["affect_fit"] is not None:
        parts.append(
            "lively videos land in its loud parts and calm ones in its quiet parts"
            if fit["affect_fit"] >= 0.7
            else "the videos' feel does not match its loud and quiet parts well"
        )
    if fit["uncovered_s"] > 0:
        parts.append(f"{fit['uncovered_s']:.0f} s of it would have no picture")
    parts.append(
        f"the footage would be shown about {fit['times_over']:.1f} times over to fill it"
        if fit["times_over"] > 1.05
        else "it needs no footage twice"
    )
    return f"{name}: " + "; ".join(parts) + "."


def _section_energy(analysis) -> Callable[[float], float]:
    """Song time -> the loudness of its section in [-1, 1] (0 when unmeasured)."""
    secs = [s for s in analysis.sections if s.energy_db is not None]
    if len(secs) < 2:
        return lambda t: 0.0
    dbs = np.array([s.energy_db for s in secs])
    mid, half = float(np.mean(dbs)), float(np.ptp(dbs)) / 2.0 or 1.0

    def energy(t: float) -> float:
        for s in secs:
            if s.start <= t < s.end:
                return max(-1.0, min(1.0, (s.energy_db - mid) / half))
        return 0.0

    return energy

def fill_spans(
    song_path: str,
    sources: Sequence[FreeSource],
    *,
    spans: Optional[Sequence[tuple[float, float]]] = None,
    song_duration: Optional[float] = None,
    analysis: Any = None,
    archetype: str = ARCHETYPE_AUTO,
    cut_feel: str = "",
    envelope_of: Optional[Callable[[FreeSource], "Optional[Envelope]"]] = None,
    anchor_of: Callable[[FreeSource], tuple[float, float]] = _default_anchor,
    canvas: tuple[int, int] = (1920, 1080),
    fps: float = 30.0,
    fit: str = "cover",
    size_of: Callable[[str], Optional[tuple[int, int]]] = display_size,
    order: str = DEFAULT_ORDER,
    affect_of: Optional[
        Callable[[FreeSource, "Optional[Envelope]"], Optional[float]]
    ] = liveliness,
    max_cuts: Optional[int] = None,
) -> MusicCut:
    """Free cuts of ``sources`` covering ``spans`` of the song (default: all of it).

    Seams, each with a default that works: ``analysis`` (the song's beats, bars and
    sections — :func:`muvid.montage.analysis.analyze`), ``envelope_of`` (a video's
    picture-change :class:`Envelope`; ``None`` falls back to the planner's blind stride),
    ``anchor_of`` (a still's focal point). ``archetype`` is a montage archetype with one
    picture per slot, or ``"auto"``; ``cut_feel`` a montage pace (``slow``, ``steady``,
    ``driving``, ``frantic``). ``fit`` is one of :data:`FITS`; ``size_of`` reads a
    picture's displayed size (for ``cover``). ``order`` is one of :data:`ORDERS`: by
    default the pictures follow the order they were TAKEN in
    (:func:`chronological_picks`); the report says which order applied and why.
    ``affect_of`` (muvid#134) gives each picture's feel — ``(source, envelope) ->`` a
    number, higher livelier, or ``None`` — and the planner then puts lively pictures in
    the loud sections and calm ones in the quiet sections (default :func:`liveliness`;
    ``None`` turns it off). It reorders pictures, so it acts in the ``planner`` order
    only; ``taken`` keeps the order they were taken in. The report's ``affect`` says
    whether it applied. ``max_cuts`` (default :data:`~muvid.footage.edl.MAX_EDL_ENTRIES`)
    is the most cuts the plan may make: past it, consecutive slots of a section are held
    as one picture (:func:`_hold_slots`) until it fits, and the report's ``held`` and
    ``cap_note`` say so (muvid#228) — so the planner never proposes an edit the gate is
    certain to refuse.

    Returns :class:`MusicCut` whose ``entries`` are validated-shape
    :class:`~muvid.footage.edl.EdlEntry` free cuts, contiguous over each span, in song
    order (spans with nothing in them are simply not covered — the caller's
    ``fill_gaps`` makes that explicit).
    """
    from muvid.montage.plan import plan_montage

    if not sources:
        raise ValueError("no footage to cut to the music")
    if analysis is None:
        analysis = _default_analysis(song_path)
    duration = float(song_duration or analysis.duration)
    analysis, grid_note = extend_grid(analysis, duration)
    spans = (
        [(0.0, duration)] if spans is None else [(float(a), float(b)) for a, b in spans]
    )
    spans = [(a, b) for a, b in spans if b - a > MIN_PIECE_S]
    chosen = (
        pick_archetype(analysis) if archetype in ("", ARCHETYPE_AUTO) else archetype
    )
    if chosen not in SINGLE_PICTURE_ARCHETYPES:
        raise ValueError(
            f"archetype {chosen!r} cannot be an edit (one picture per cut); use one of "
            f"{list(SINGLE_PICTURE_ARCHETYPES)} or 'auto'"
        )
    treatment = _treatment(chosen, cut_feel)
    if order not in ORDERS:
        raise ValueError(f"order must be one of {list(ORDERS)}, got {order!r}")
    media = _media_for(sources)
    if order == "taken":
        from dataclasses import replace as _replace

        direction = treatment.direction
        treatment = _replace(
            treatment,
            direction=_replace(
                direction, reuse=_replace(direction.reuse, order="chronological")
            ),
        )
    envelopes: dict[str, Optional[Envelope]] = {}

    def envelope(src: FreeSource) -> "Optional[Envelope]":
        if src.clip_id not in envelopes:
            envelopes[src.clip_id] = (
                None if envelope_of is None or src.is_still else envelope_of(src)
            )
        return envelopes[src.clip_id]

    energy_at = _section_energy(analysis)
    mismatch = (
        None
        if order == "taken"
        else _affect_mismatch(
            sources, analysis, energy_at, affect_of=affect_of, envelope=envelope
        )
    )
    affect_fit = (  # the planner's shape: (slot, media) -> mismatch
        None
        if mismatch is None
        else (lambda draft, media: mismatch(draft.start, draft.end, media.index))
    )
    plan = plan_montage(
        analysis,
        media,
        treatment,
        active=lambda d: any(
            d.start < b - _SAME_FLOAT and d.end > a + _SAME_FLOAT for a, b in spans
        ),
        affect_fit=affect_fit,
    )
    order_used = plan.reuse.get("order", "planner")
    order_note = plan.reuse.get("order_note", "") or _mixed_clock_note(sources)
    beats = np.asarray(analysis.beats, dtype=float)
    if fit not in FITS:
        raise ValueError(f"fit must be one of {list(FITS)}, got {fit!r}")
    crops: dict[str, Any] = {}

    def crop_for(src: FreeSource):
        if fit != "cover":
            return None
        if src.clip_id not in crops:
            anchor = (
                anchors.setdefault(src.clip_id, anchor_of(src))
                if src.is_still
                else (0.5, 0.5)
            )
            crops[src.clip_id] = cover_crop(size_of(src.path), canvas, anchor=anchor)
        return crops[src.clip_id]

    anchors: dict[str, tuple[float, float]] = {}

    def cut(slots):
        """The plan's ``slots`` as free cuts: ``(entries, uses, fits)``."""
        used: dict[str, list[tuple[float, float]]] = {}
        fits: dict[str, list[float]] = {}
        uses: dict[str, int] = {}
        pieces: list[dict] = []
        for slot in slots:
            for a, b in spans:
                lo, hi = max(slot.start, a), min(slot.end, b)
                if hi - lo <= _SAME_FLOAT:
                    continue
                tile = slot.tiles[0]
                fade = (
                    slot.transition_s
                    if (lo == slot.start and slot.transition != "cut")
                    else 0.0
                )
                pieces.append(
                    {
                        # a sliver a span boundary cut off a slot — not a slot the
                        # planner made short on purpose (a flip-book hold)
                        "clipped": lo > slot.start + _SAME_FLOAT
                        or hi < slot.end - _SAME_FLOAT,
                        "start": lo,
                        "end": hi,
                        "media": tile.media,
                        "motion": tile.motion,
                        "fade": fade,
                        "first_in_span": abs(lo - a) < _SAME_FLOAT,
                    }
                )
        pieces = _fold_tiny(pieces)
        _fades_start_on_the_beat(pieces)
        for this, nxt in zip(pieces, pieces[1:]):
            this["next"], nxt["prev"] = (
                {"media": nxt["media"]},
                {"media": this["media"]},
            )
            # The next cut's blend reads past this cut's end in this cut's source.
            if (
                abs(this["end"] - nxt["start"]) < _SAME_FLOAT
                and not nxt["first_in_span"]
            ):
                this["next_fade"] = nxt["fade"]

        entries = []
        for k, p in enumerate(pieces):
            entries += _cuts_for_piece(
                p,
                k,
                sources=sources,
                beats=beats,
                energy_at=energy_at,
                envelope=envelope,
                anchor_of=anchor_of,
                anchors=anchors,
                used=used,
                fits=fits,
                uses=uses,
                canvas=canvas,
                fps=fps,
                crop_for=crop_for,
            )
        entries = _feasible_blends(entries, {s.clip_id: s for s in sources})
        entries = _back_on_the_beat(
            entries,
            {round(p["start"], 6): p["shift"] for p in pieces if p.get("shift")},
            {s.clip_id: s for s in sources},
        )
        return entries, uses, fits

    from muvid.footage.edl import MAX_EDL_ENTRIES

    cap = MAX_EDL_ENTRIES if max_cuts is None else int(max_cuts)
    if cap < 1:
        raise ValueError(f"max_cuts must be at least 1, got {max_cuts}")
    held, slots = 1, plan.slots
    entries, uses, fits = cut(slots)
    # Coarsen until the plan fits the cut cap (muvid#228), as `weighted` escalates its
    # switch cost: fewer, longer pieces on the same beats. Holding never crosses a
    # section boundary, so it bottoms out — and a piece longer than its clip is still
    # split into take-overs — hence the stop when holding changes nothing more.
    while len(entries) > cap:
        coarser = _hold_slots(plan.slots, held * 2)
        if len(coarser) == len(slots):
            break
        held, slots = held * 2, coarser
        entries, uses, fits = cut(slots)
    if len(entries) > cap:
        cap_note = (
            f"{len(entries)} cuts, over the {cap}-cut limit even with each picture held "
            f"for up to {held} of the planned slots: the videos are too short to fill "
            "this song in fewer cuts — add longer videos, or cut part of the song"
        )
    elif held > 1:
        cap_note = (
            f"each picture holds up to {held} of the planned slots, to stay within the "
            f"{cap}-cut limit"
        )
    else:
        cap_note = ""
    asked = sum(b - a for a, b in spans)
    covered = sum(e.song_end - e.song_start for e in entries)
    report = {
        # seconds of the asked spans no picture could fill (left as gaps) — 0 normally
        "uncovered_s": round(max(0.0, asked - covered), 2),
        "grid_note": grid_note,
        # how many planned slots each picture holds (1 normally) and why (muvid#228)
        "held": held,
        "cap_note": cap_note,
        "tempo_bpm": round(float(analysis.tempo_bpm), 2),
        "beat_source": getattr(analysis, "beat_source", ""),
        "style": chosen,
        "pace": treatment.direction.cut_feel,
        # which order the pictures follow, and why (taken -> upload when no file says
        # when it was taken)
        "order": order_used,
        "order_note": order_note,
        # whether the pictures' feel was matched to the sections' loudness (muvid#134)
        "affect": bool(plan.reuse.get("affect")),
        # which pictures that matching kept out of which stretch of the song (muvid#234):
        # one entry per section occurrence, pictures never shown there because of it
        "affect_withheld": [
            {
                **{k: w[k] for k in ("section", "start", "end")},
                "clips": [sources[m].clip_id for m in w["media"]],
            }
            for w in plan.reuse.get("affect_withheld", [])
        ],
        # how well the pictures' feel fits the sections', over cut time: 1 a perfect
        # match, 0 the opposite; None when affect did not apply (muvid#135 ranks on it)
        "affect_fit": _cut_affect_fit(entries, sources, mismatch),
        "sections": [
            {"label": s.label, "start": round(s.start, 2), "end": round(s.end, 2)}
            for s in analysis.sections
        ],
        "n_cuts": len(entries),
        "uses": {cid: n for cid, n in sorted(uses.items())},
        # Mean "hits on the beat" per video: above 1, its picture changes on the beat
        # more than it does anywhere else in it.
        "on_beat": {
            cid: round(float(np.mean(v)), 3) for cid, v in sorted(fits.items()) if v
        },
        "measured_pictures": sorted(c for c, e in envelopes.items() if e is not None),
    }
    return MusicCut(entries=entries, report=report)




def _hold_slots(slots, every: int) -> list:
    """``slots`` with each run of up to ``every`` consecutive slots OF ONE SECTION held
    as its first slot's picture (muvid#228): the same beats, fewer cuts. A section's
    first slot keeps its own boundary, so a cut still lands where the music changes.

    >>> from muvid.montage.plan import Slot
    >>> s = [Slot(index=i, start=i, end=i + 1, section="v" if i < 3 else "c",
    ...           archetype="x", regions=1, tiles=()) for i in range(5)]
    >>> [(x.start, x.end, x.section) for x in _hold_slots(s, 2)]
    [(0, 2, 'v'), (2, 3, 'v'), (3, 5, 'c')]
    """
    from dataclasses import replace as _replace

    out, run = [], 0
    for slot in slots:
        if out and run < every and slot.section == out[-1].section:
            out[-1] = _replace(out[-1], end=slot.end)
            run += 1
        else:
            out.append(slot)
            run = 1
    return out


def _fades_start_on_the_beat(pieces: list[dict]) -> None:
    """Move each faded boundary later by the fade's lead, so the fade STARTS on the
    beat the planner chose, instead of straddling it (measured on a real edit: 44 of
    62 cuts were fades centred on their beat, so the picture began changing a third
    of a second early — heard as "slightly off the beat"). In place; a piece the move
    would leave shorter than :data:`MIN_PIECE_S` keeps a hard cut instead."""
    from muvid.footage.edl import Transition, blend_lead

    for prev, p in zip(pieces, pieces[1:]):
        fade = p.get("fade") or 0.0
        if (
            fade <= 0
            or p["first_in_span"]
            or abs(prev["end"] - p["start"]) > _SAME_FLOAT
        ):
            continue
        lead = blend_lead(Transition(duration_s=fade, curve="fade"))
        if p["end"] - (p["start"] + lead) < MIN_PIECE_S:
            p["fade"] = 0.0
            continue
        prev["end"] = p["start"] = p["start"] + lead
        p["shift"] = lead  # undone if the blend is later dropped (_back_on_the_beat)


def _fold_tiny(pieces: list[dict]) -> list[dict]:
    """Fold the slivers a span boundary clipped off a slot — shorter than
    :data:`MIN_PIECE_S` — into the previous piece of the same span (or the next
    one), so a clipped slot leaves no flash frames. A slot the planner made short
    ON PURPOSE (a flip-book hold of a third of a second) is left alone: folding
    those cascades them into one long hold."""
    out: list[dict] = []
    for p in pieces:
        short = p.get("clipped", True) and p["end"] - p["start"] < MIN_PIECE_S
        if (
            short
            and out
            and not p["first_in_span"]
            and abs(out[-1]["end"] - p["start"]) < _SAME_FLOAT
        ):
            out[-1] = dict(out[-1], end=p["end"])
        else:
            out.append(dict(p))
    merged: list[dict] = []
    for p in out:
        if (
            merged
            and merged[-1].get("clipped", True)
            and merged[-1]["end"] - merged[-1]["start"] < MIN_PIECE_S
            and abs(merged[-1]["end"] - p["start"]) < _SAME_FLOAT
            and not p["first_in_span"]
        ):
            p = dict(p, start=merged[-1]["start"], fade=0.0)
            merged[-1] = p
        else:
            merged.append(p)
    return merged


def _cuts_for_piece(
    p,
    k,
    *,
    sources,
    beats,
    energy_at,
    envelope,
    anchor_of,
    anchors,
    used,
    fits,
    uses,
    canvas,
    fps,
    crop_for,
) -> list:
    """One planned piece -> one or more free cuts (more when its video is shorter than
    the piece: the rest goes to the longest other source)."""
    from muvid.footage.edl import COINCIDENT_S, EdlEntry, Transition, longest_span

    def fade_of(seconds):
        return Transition(duration_s=seconds, curve="fade") if seconds > 0 else None

    out = []
    start, end = p["start"], p["end"]
    src = sources[p["media"]]
    # the pictures either side of this piece, so a take-over does not repeat them
    avoid = {sources[q["media"]].clip_id for q in (p.get("prev"), p.get("next")) if q}
    fade = p["fade"] if not p["first_in_span"] else 0.0
    next_blend = fade_of(p.get("next_fade", 0.0))  # the blend out of this piece
    steps, max_steps = 0, int((end - start) / MIN_PIECE_S) + 4
    while end - start > COINCIDENT_S and steps < max_steps:
        steps += 1
        no_trail = False
        length = end - start
        own = fade_of(fade)
        holds = longest_span(
            src.duration_s or 0.0,
            transition=own,
            next_transition=next_blend,
            edge_s=EDGE_S,
        )
        if not src.is_still and holds < length - _SAME_FLOAT:
            # The video cannot hold the whole piece: show what it has, ending on a beat
            # before the piece's end, so a take-over continues it with a hard cut and
            # this cut carries no next blend. Ending ON the piece's end would carry it
            # — which `holds` just said does not fit — so that end is capped by `holds`,
            # and no earlier than leaves a real piece after it. Where no such cut point
            # exists the whole piece stays, without reserving the next blend's trail:
            # `_feasible_blends` then turns that blend into a hard cut if the gate
            # refuses it, and where it still fits the edge margin gives way instead —
            # coverage is never what is traded (the #203 review).
            cap = longest_span(src.duration_s or 0.0, transition=own, edge_s=EDGE_S)
            cut_at = _last_beat_before(beats, start + cap, after=start + MIN_PIECE_S)
            if cut_at is not None and cut_at >= end - COINCIDENT_S:
                cut_at = _last_beat_before(
                    beats,
                    min(start + holds, end - MIN_PIECE_S),
                    after=start + MIN_PIECE_S,
                )
                if cut_at is None:
                    cut_at, no_trail = end, True
            if cut_at is None or cap < MIN_PIECE_S:
                nxt = _take_over(
                    sources,
                    src,
                    need=length,
                    avoid=avoid,
                    uses=uses,
                    transition=own,
                    next_transition=next_blend,
                )
                if nxt.clip_id == src.clip_id:
                    break  # nothing else to show: the rest stays uncovered (reported)
                avoid.add(src.clip_id)
                src = nxt
                continue
            piece_end = min(cut_at, end)
        else:
            piece_end = end
        trail = p.get("next_fade", 0.0) if piece_end == end and not no_trail else 0.0
        entry = _one_cut(
            src,
            start,
            piece_end,
            fade=fade,
            trail=trail,
            motion=p["motion"],
            k=k,
            beats=beats,
            energy_at=energy_at,
            envelope=envelope,
            anchor_of=anchor_of,
            anchors=anchors,
            used=used,
            fits=fits,
            uses=uses,
            canvas=canvas,
            fps=fps,
            transition_cls=Transition,
            entry_cls=EdlEntry,
        )
        crop = crop_for(src)
        if crop is not None:
            from dataclasses import replace as _replace

            entry = _replace(entry, crop=crop)
        out.append(entry)
        start, fade = piece_end, 0.0
        if end - start > COINCIDENT_S:
            avoid.add(src.clip_id)
            # the take-over starts on a hard cut and may end carrying the next blend
            src = _take_over(
                sources,
                src,
                need=end - start,
                avoid=avoid,
                uses=uses,
                next_transition=next_blend,
            )
    return out


def _last_beat_before(beats, t: float, *, after: float) -> Optional[float]:
    cands = beats[(beats <= t + _SAME_FLOAT) & (beats >= after)]
    if cands.size:
        return float(cands[-1])
    return t if t >= after else None


def _take_over(
    sources,
    current,
    *,
    need: float,
    avoid: set,
    uses: Mapping[str, int],
    transition=None,
    next_transition=None,
):
    """A source to take over the rest of a piece: one that can hold ``need`` seconds (a
    still always can) — preferably not a picture either side, the least used first —
    and only when nothing can hold it, the longest of the rest. ``current`` back means
    there is nothing else at all.

    "Hold" counts the blends the piece carries — ``transition`` in, ``next_transition``
    out — by the same rule :func:`_cuts_for_piece` then applies (``longest_span``), so a
    clip that holds ``need`` only without its blends is not picked just to be found
    short and replaced on the next pass (muvid#210)."""
    others = [s for s in sources if s.clip_id != current.clip_id]
    if not others:
        return current

    from muvid.footage.edl import longest_span

    def holds(s):
        return (
            s.is_still
            or longest_span(
                s.duration_s or 0.0,
                transition=transition,
                next_transition=next_transition,
                edge_s=EDGE_S,
            )
            >= need
        )

    able = [s for s in others if holds(s)]
    pool = [s for s in able if s.clip_id not in avoid] or able
    if pool:
        return min(
            pool,
            key=lambda s: (uses.get(s.clip_id, 0), -(s.duration_s or 1e9), s.clip_id),
        )
    fresh = [s for s in others if s.clip_id not in avoid] or others
    return max(fresh, key=lambda s: (s.is_still, s.duration_s or 0.0))


def _one_cut(
    src,
    start,
    end,
    *,
    fade,
    trail,
    motion,
    k,
    beats,
    energy_at,
    envelope,
    anchor_of,
    anchors,
    used,
    fits,
    uses,
    canvas,
    fps,
    transition_cls,
    entry_cls,
):
    from dataclasses import replace

    from muvid.footage.edl import still_in_point
    from muvid.footage.named_looks import compile_named_look, resolve_named_look

    length = end - start
    transition = transition_cls(duration_s=fade, curve="fade") if fade > 0 else None
    # the FOLLOWING cut's blend, which reads `trail` past this cut's end
    next_transition = (
        transition_cls(duration_s=trail, curve="fade") if trail > 0 else None
    )
    draft = entry_cls(
        song_start=start, song_end=end, clip_id=src.clip_id, transition=transition
    )
    use = uses.get(src.clip_id, 0)
    uses[src.clip_id] = use + 1
    if src.is_still:
        name = MOTION_TO_LOOK.get(motion) or STILL_MOVES[use % len(STILL_MOVES)]
        spec: dict = {"name": name, "zoom": STILL_ZOOM}
        if name in _ANCHORED_LOOKS:
            if src.clip_id not in anchors:
                anchors[src.clip_id] = anchor_of(src)
            ax, ay = anchors[src.clip_id]
            spec |= {"anchor_x": round(ax, 3), "anchor_y": round(ay, 3)}
        resolved = resolve_named_look(spec)
        look = compile_named_look(resolved, canvas=canvas, fps=fps, duration_s=length)
        return replace(
            draft,
            look=look,
            look_time_varying=bool(getattr(look, "time_varying", True)),
            look_spec=resolved,
            # a blend reads its lead BEFORE the cut's in-point: the gate's own lower
            # bound (a still has no thumb on the button, so no edge)
            source_in=round(still_in_point(draft), 4),
        )
    inner = [float(b - start) for b in beats if start < b < end]
    clip_duration = float(src.duration_s or length)
    s_in, fit = choose_source_in(
        envelope(src),
        clip_duration=clip_duration,
        length=length,
        beat_offsets=inner,
        used=used.get(src.clip_id, ()),
        energy=energy_at((start + end) / 2.0),
        room=_room(draft, clip_duration=clip_duration, next_transition=next_transition),
        ordinal=k,
    )
    used.setdefault(src.clip_id, []).append((s_in, s_in + length))
    if inner:
        fits.setdefault(src.clip_id, []).append(fit)
    return replace(draft, source_in=round(s_in, 4))


def _back_on_the_beat(entries: list, shifted: Mapping[float, float], by_id) -> list:
    """A boundary moved later so its fade would START on the beat, whose fade was
    then dropped (no footage to blend), goes back ON the beat as a hard cut — else
    the fallback lands a third of a second after it. Skipped where the move back
    would read past either clip's footage."""
    from dataclasses import replace

    from muvid.footage.edl import with_start

    out = list(entries)
    for i in range(1, len(out)):
        e, prev = out[i], out[i - 1]
        shift = shifted.get(round(e.song_start, 6))
        if (
            not shift
            or e.transition is not None
            or abs(prev.song_end - e.song_start) > _SAME_FLOAT
        ):
            continue
        at = e.song_start - shift
        if at - prev.song_start < MIN_PIECE_S:
            continue
        moved = (
            with_start(e, at, bounded=False)
            if e.source_in is not None
            else replace(e, song_start=at)
        )
        src = by_id.get(e.clip_id)
        if moved.source_in is not None and moved.source_in < 0:
            continue
        if src is not None and not src.is_still and moved.source_in is not None:
            if (
                moved.source_in + (moved.song_end - at) * moved.rate
                > (src.duration_s or 0.0) + _SAME_FLOAT
            ):
                continue
        out[i - 1], out[i] = replace(prev, song_end=at), moved
    return out


def _feasible_blends(entries: list, by_id: Mapping[str, FreeSource]) -> list:
    """Drop a blend the edit gate would refuse — a short video that could not keep the
    margin, a cut too short to give its neighbour the blend — asking the gate's own
    check (:func:`muvid.footage.edl._validate_transition`), so the two cannot drift. A
    hard cut there is the honest fallback."""
    from dataclasses import replace

    from muvid.footage.edl import STILL_DURATION_S, _validate_transition, unplaced

    places = {
        cid: unplaced(
            cid, STILL_DURATION_S if s.is_still else float(s.duration_s or 0.0)
        )
        for cid, s in by_id.items()
    }
    out = list(entries)
    for i in range(len(out)):
        e = out[i]
        if e.transition is None:
            continue
        prev = out[i - 1] if i > 0 else None
        try:
            if prev is None or abs(prev.song_end - e.song_start) > _SAME_FLOAT:
                raise ValueError("no contiguous predecessor")
            _validate_transition(i, e, prev, places)
        except (ValueError, KeyError):
            out[i] = replace(e, transition=None)
    return out
